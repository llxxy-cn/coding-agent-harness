> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Build a fixed, offline, deterministic demo Web UI exposing three scenarios through server-rendered pages, reusing existing HarnessCore and ApplicationService.

**Architecture:** FastAPI app factory (`create_demo_app`) accepts `WebSettings`, `ScenarioRegistry`, `DemoRunService` (keyword-required). ## Global Constraints

- Python >=3.11,<3.13; all commands use `py -3.12` (NOT `python` which is 3.14).
- All new Pydantic models: `ConfigDict(frozen=True, extra="forbid", use_enum_values=False)`.
- No real Provider, keyring, or Git remote. Offline demo only.
- No baseline pytest in scenarios. Typed trace (enum/bounded int/bool only). Fail-closed.
- Cleanup failure overrides success: 500 `cleanup_failed`, no trace.
- `DemoFeedbackCode` (not domain `FeedbackKind`) with `INITIAL_FAILURE` for `kind=None` path.
- Page text from fixed enum codes. No approve/reject/resume endpoint. Single process/worker, concurrency 1.
- `MANIFEST.in`/`pyproject.toml` changes proven by Red test.
- `ParsedOutcome` enum (in `feedback/pytest_parser.py:16-21`): PASSED, FAILED, COLLECTION_ERROR, ENVIRONMENT_ERROR, UNPARSEABLE — NO `TIMED_OUT`. Do not confuse with `TestRunOutcome` (in `domain/enums.py`) which has `TIMED_OUT`.
- Error priority uses explicit integer rank (0=termination_unconfirmed, 1=cleanup_failed, 2=timeout/trace/internal, 3=success). Lower rank = higher priority. NEVER use string/Enum alphabetical comparison.
- State machine: NO `return` in `finally`. Use `pending_result` variable. Finally order: close file handles → confirm/terminate process → close process/Job handles → compute safe_to_cleanup → cleanup/quarantine → release lock (always last) → generate response once at function end.

**Key logic — sequential event mapping (FIX for Block 2):**

```python
def run_scenario(scenario_id, tmp_path) -> tuple:
    """Copy template, run build_demo_runtime, inspect session, build events."""
    # ... setup ...
    view = runtime.run(repository=repo, task_description=cfg.task_description, mode="demo", trust_repo=True)
    session = runtime.runtime.application.session_store.load(view.task_id)
    events = _build_trace_events(session)
    return events, view, session

def _build_trace_events(session) -> tuple:
    """Process history SEQUENTIALLY — emit test+feedback pairs, not batched."""
    events = []
    seq = 0
    prev_test_outcome = None  # Track for feedback determination
    for entry in session.history:
        if entry.action_type == "apply_patch":
            events.append(ActionSelectedEvent(action_kind=DemoActionKind.APPLY_PATCH, sequence=seq)); seq += 1
        elif entry.action_type == "request_human":
            events.append(ActionSelectedEvent(action_kind=DemoActionKind.REQUEST_HUMAN, sequence=seq)); seq += 1
        elif entry.action_type == "full_test":
            # Determine outcome from safe_result — use ParsedOutcome, NOT TestRunOutcome
            # ParsedOutcome: PASSED, FAILED, COLLECTION_ERROR, ENVIRONMENT_ERROR, UNPARSEABLE
            outcome = TestRunOutcome.FAILED if "fail" in entry.safe_result.lower() else TestRunOutcome.PASSED
            events.append(TestCompletedEvent(outcome=outcome, sequence=seq)); seq += 1
            # IMMEDIATELY emit feedback (not batched)
            if prev_test_outcome is None and outcome is TestRunOutcome.FAILED:
                events.append(FeedbackProducedEvent(feedback_code=DemoFeedbackCode.INITIAL_FAILURE, sequence=seq)); seq += 1
            elif outcome is TestRunOutcome.PASSED:
                events.append(FeedbackProducedEvent(feedback_code=DemoFeedbackCode.PASSED, sequence=seq)); seq += 1
            else:
                events.append(FeedbackProducedEvent(feedback_code=DemoFeedbackCode.CHANGED, sequence=seq)); seq += 1
            prev_test_outcome = outcome
        elif entry.action_type == "policy_deny":
            events.append(PolicyDecisionEvent(decision=PolicyOutcome.DENY, reason_code=PolicyReasonCode.TEST_ASSET_PROTECTION, sequence=seq)); seq += 1
    events.append(TerminalStatusEvent(task_status=session.status, sequence=seq))
    return tuple(events)
```

**Critical:** The `while` loop processes entries one at a time. After each `full_test`, it immediately emits `TestCompletedEvent` then `FeedbackProducedEvent`. This produces the correct 7-event sequence for feedback_success: `action(0)→test(1)→feedback(2)→action(3)→test(4)→feedback(5)→terminal(6)`.

**Note on ParsedOutcome (FIX for Block 1):** The helper uses `TestRunOutcome` (from `domain/enums.py`) for `TestCompletedEvent.outcome` because that's what the event type requires. The helper does NOT reference `ParsedOutcome.TIMED_OUT` (which doesn't exist). Outcome is determined from `entry.safe_result` string content, not from `ParsedOutcome` enum.

def run_scenario(self, scenario_id, request_root_parent) -> ExecutionTraceView:
    pending_result = RunResult(outcome="success")  # default
    worker_handle = None
    request_root = None
    lock_acquired = False
    try:
        if not self._try_acquire_lock():
            return RunResult(outcome="busy")  # 503
        lock_acquired = True
        request_root = Path(request_root_parent) / "request_root"
        request_root.mkdir(); (request_root/"data").mkdir(); (request_root/"cwd").mkdir()
        # Write worker-config.json
        worker_handle = self._start_worker(config_path, result_path, request_root/"cwd")
        deadline = self._monotonic() + 60
        exit_code = worker_handle.wait(deadline)
        if exit_code is None:  # timeout
            worker_handle.terminate_tree()
            if not worker_handle.confirm_termination():
                pending_result = RunResult(outcome="termination_unconfirmed")
            else:
                pending_result = RunResult(outcome="timeout")
        else:  # normal exit
            if not worker_handle.confirm_termination():  # descendants alive
                worker_handle.terminate_tree()
                if not worker_handle.confirm_termination():
                    pending_result = RunResult(outcome="termination_unconfirmed")
                else:
                    pending_result = RunResult(outcome="worker_lifecycle_invalid")
            else:
                # Collect and validate result
                result = self._read_and_validate_result(result_path, scenario_id, cfg)
                if isinstance(result, str):
                    pending_result = RunResult(outcome=result)
                else:
                    try:
                        cfg.trace_contract.validate(result.events)
                    except ValueError:
                        pending_result = RunResult(outcome="trace_incomplete")
                    else:
                        pending_result = RunResult(outcome="success", view=ExecutionTraceView.from_events(result.events))
    except Exception:
        if _ERROR_RANK["internal"] < _ERROR_RANK[pending_result.outcome]:
            pending_result = RunResult(outcome="internal")
    finally:
        # 1. Close file handles (if any open)
        # 2. Confirm/terminate process boundary
        if worker_handle is not None:
            if not worker_handle.confirm_termination():
                worker_handle.terminate_tree()
                if not worker_handle.confirm_termination():
                    if _ERROR_RANK["termination_unconfirmed"] < _ERROR_RANK[pending_result.outcome]:
                        pending_result = RunResult(outcome="termination_unconfirmed")
            worker_handle.close()  # 3. Close process/Job handles
        # 4. Compute safe_to_cleanup
        safe_to_cleanup = request_root is not None and (worker_handle is None or worker_handle.confirm_termination())
        # 5. Cleanup/quarantine
        if request_root is not None:
            if safe_to_cleanup:
                try:
                    shutil.rmtree(request_root)
                except Exception:
                    if _ERROR_RANK["cleanup_failed"] < _ERROR_RANK[pending_result.outcome]:
                        pending_result = RunResult(outcome="cleanup_failed")
            else:
                if _ERROR_RANK["termination_unconfirmed"] < _ERROR_RANK[pending_result.outcome]:
                    pending_result = RunResult(outcome="termination_unconfirmed")
        # 6. Release lock (ALWAYS last, NO return before this)
        if lock_acquired:
            self._lock.release()
    # 7. Generate response once at function end
    if pending_result.outcome != "success":
        raise RunError(pending_result.outcome)
    return pending_result.view
```

# web/process_boundary.py
import ctypes
import sys
import subprocess
from typing import Protocol

class ProcessBoundary(Protocol):
    def wait(self, deadline: float) -> int | None: ...
    def confirm_termination(self) -> bool: ...
    def terminate_tree(self) -> None: ...
    def close(self) -> None: ...

class WindowsJobObjectBoundary:
    """Real Windows Job Object with kill-on-close."""
    def __init__(self, process: subprocess.Popen):
        self._process = process
        self._job = ctypes.windll.kernel32.CreateJobObjectW(None, None)
        # Set JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE
        info = JOBOBJECT_EXTENDED_LIMIT_INFORMATION()
        info.BasicLimitInformation.LimitFlags = 0x2000  # JOB_OBJECT_LIMIT_KILL_ON_JOB_CLOSE
        ctypes.windll.kernel32.SetInformationJobObject(self._job, 9, ctypes.byref(info), ctypes.sizeof(info))
        ctypes.windll.kernel32.AssignProcessToJobObject(self._job, self._process._handle)
        # Resume if CREATE_SUSPENDED was used
        ctypes.windll.kernel32.ResumeThread(self._process._handle)

    def wait(self, deadline: float) -> int | None:
        timeout = max(0, deadline - time.monotonic())
        try: return self._process.wait(timeout=timeout)
        except subprocess.TimeoutExpired: return None

    def confirm_termination(self) -> bool:
        # QueryInformationJobObject for active process count
        buf = (ctypes.c_ulong * 4)()
        ret = ctypes.windll.kernel32.QueryInformationJobObject(self._job, 5, buf, ctypes.sizeof(buf), None)
        return buf[0] == 0  # active_count == 0

    def terminate_tree(self) -> None:
        ctypes.windll.kernel32.CloseHandle(self._job)  # kill-on-close

    def close(self) -> None:
        if self._job: ctypes.windll.kernel32.CloseHandle(self._job)
        if self._process.stdout: self._process.stdout.close()
        if self._process.stderr: self._process.stderr.close()

class PosixProcessBoundary:
    """Real POSIX process-group termination."""
    def __init__(self, process: subprocess.Popen):
        self._process = process
        self._pgid = os.getpgid(process.pid)

    def wait(self, deadline: float) -> int | None:
        timeout = max(0, deadline - time.monotonic())
        try: return self._process.wait(timeout=timeout)
        except subprocess.TimeoutExpired: return None

    def confirm_termination(self) -> bool:
        try:
            os.killpg(self._pgid, 0)  # check if group exists
            return False  # still alive
        except ProcessLookupError:
            return True  # gone

    def terminate_tree(self) -> None:
        try: os.killpg(self._pgid, signal.SIGKILL)
        except ProcessLookupError: pass

    def close(self) -> None:
        if self._process.stdout: self._process.stdout.close()
        if self._process.stderr: self._process.stderr.close()

class FakeProcessBoundary:
    """Test injection — no real subprocess."""
    def __init__(self, *, wait_return=0, confirm=True):
        self._wait_return = wait_return
        self._confirm = confirm
    def wait(self, deadline): return self._wait_return
    def confirm_termination(self): return self._confirm
    def terminate_tree(self): pass
    def close(self): pass

def create_boundary(cmd: list, *, cwd, env, creationflags=None, start_new_session=False) -> ProcessBoundary:
    """Factory: select platform-appropriate boundary."""
    proc = subprocess.Popen(cmd, shell=False, cwd=cwd, env=env,
                           stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                           creationflags=creationflags, start_new_session=start_new_session)
    if sys.platform == "win32":
        return WindowsJobObjectBoundary(proc)
    return PosixProcessBoundary(proc)
```
